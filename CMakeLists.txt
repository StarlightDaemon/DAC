cmake_minimum_required(VERSION 3.25)
# Embedded debug information avoids a shared compiler PDB-server lifetime and
# keeps parallel/isolated builds self-contained (CMP0141 is NEW in CMake 3.25).
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
project(DAC VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "DAC currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(DAC_BUILD_TESTS "Build isolated automated test fixtures" ON)
include(cmake/Fujin.cmake)

add_library(dac_options INTERFACE)
target_compile_definitions(dac_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_include_directories(dac_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/resources")
if(MSVC)
  target_compile_options(dac_options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc /guard:cf /sdl)
  target_link_options(dac_options INTERFACE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf /INCREMENTAL:NO /Brepro)
else()
  target_compile_options(dac_options INTERFACE -Wall -Wextra -Werror -Wno-unknown-pragmas -fstack-protector-strong)
  target_link_options(dac_options INTERFACE -static -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
set(DAC_WINDOWS_LIBS shell32 user32 gdi32 ole32 uuid wtsapi32 comdlg32 advapi32 powrprof dwmapi dxva2 gdiplus)
function(dac_native target)
  target_link_libraries(${target} PRIVATE dac_options ${DAC_WINDOWS_LIBS})
  dac_add_fujin(${target})
endfunction()

add_executable(DAC WIN32 src/standalone.cpp resources/dac.rc)
dac_native(DAC)
if(MINGW)
  target_link_options(DAC PRIVATE -municode)
endif()
if(MSVC)
  # The resource owns the explicit asInvoker/PMv2/long-path manifest.
  target_link_options(DAC PRIVATE /MANIFEST:NO)
endif()

if(DAC_BUILD_TESTS)
  enable_testing()
  add_executable(policy-tests tests/policy.cpp)
  target_link_libraries(policy-tests PRIVATE dac_options)
  add_executable(platform-tests tests/platform.cpp)
  dac_native(platform-tests)
  target_link_libraries(platform-tests PRIVATE psapi)
  add_executable(wake-probe tests/nativewake-negative.cpp)
  dac_native(wake-probe)
  add_executable(dac-test-runner tests/test-runner.cpp)
  target_link_libraries(dac-test-runner PRIVATE dac_options user32)
  add_executable(host-lifecycle-tests tests/host-lifecycle.cpp)
  target_link_libraries(host-lifecycle-tests PRIVATE dac_options user32 psapi)
  foreach(target platform-tests wake-probe dac-test-runner host-lifecycle-tests)
    if(MINGW)
      target_link_options(${target} PRIVATE -municode)
    endif()
  endforeach()
  add_test(NAME policy COMMAND policy-tests)
  add_test(NAME queued-wake-probe COMMAND dac-test-runner $<TARGET_FILE:wake-probe>)
  foreach(check storage catalog faults containment host-death emergency lifecycle sessions session-soak slides power advanced nextbeta nativewake wakeperf idleperf media host-preferences)
    add_test(NAME ${check} COMMAND dac-test-runner $<TARGET_FILE:platform-tests> --${check})
    set_tests_properties(${check} PROPERTIES TIMEOUT 130 RUN_SERIAL TRUE)
  endforeach()
  set_tests_properties(queued-wake-probe PROPERTIES TIMEOUT 130 RUN_SERIAL TRUE)
  add_test(NAME standalone-cli-version COMMAND dac-test-runner $<TARGET_FILE:DAC> --version)
  set_tests_properties(standalone-cli-version PROPERTIES PASS_REGULAR_EXPRESSION "DAC 0.1.0" TIMEOUT 10)

  add_test(NAME hardware-helper-reject COMMAND dac-test-runner --expect-exit 10 $<TARGET_FILE:DAC> --power-helper 41 aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa 4294967295)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/isolated-profile")

  set_tests_properties(standalone-lifecycle PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
  add_test(NAME fujin-integrity COMMAND "${DAC_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-theme.ps1" -BuildDirectory "${CMAKE_CURRENT_BINARY_DIR}" -SourceDirectory "${FUJIN_SOURCE_DIR}")

# Share the reviewed data-only allowlist with the portable package script.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tools/package-manifest.json" DAC_PACKAGE_MANIFEST)
string(JSON DAC_PACKAGE_FILE_COUNT LENGTH "${DAC_PACKAGE_MANIFEST}" files)
math(EXPR DAC_PACKAGE_LAST_FILE "${DAC_PACKAGE_FILE_COUNT} - 1")
install(TARGETS DAC RUNTIME DESTINATION .)
foreach(index RANGE 0 ${DAC_PACKAGE_LAST_FILE})
  string(JSON kind GET "${DAC_PACKAGE_MANIFEST}" files ${index} kind)
  if(kind STREQUAL "repository")
    string(JSON source GET "${DAC_PACKAGE_MANIFEST}" files ${index} source)
    string(JSON path GET "${DAC_PACKAGE_MANIFEST}" files ${index} path)
    get_filename_component(destination "${path}" DIRECTORY)
    get_filename_component(name "${path}" NAME)
    if(destination STREQUAL "")
      set(destination ".")
    endif()
    install(FILES "${source}" DESTINATION "${destination}" RENAME "${name}")
  endif()
endforeach()
